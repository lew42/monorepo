// Runs the real task_for source from Server/health.mjs (extracted, not copied by hand).
import fs from "fs"; import path from "path";
const ROOT = process.cwd(), PUBLIC = path.join(ROOT, "public"), AI_DIR = path.join(PUBLIC, "framework", "ai");
const DAY_MS = 86400000;
const local_date = (d = new Date()) => { const z = new Date(d - d.getTimezoneOffset() * 60000); return z.toISOString().slice(0, 10); };
const src = fs.readFileSync(path.join(ROOT, "Server/health.mjs"), "utf8");
const body = src.slice(src.indexOf("const read_lines"), src.indexOf("function summary("));
const task_for = new Function("fs", "path", "ROOT", "AI_DIR", "DAY_MS", "local_date", body + "\nreturn task_for;")(fs, path, ROOT, AI_DIR, DAY_MS, local_date);

const old_dir = path.join(AI_DIR, "2026-09-25", "health-taskfor-proof");
const landed_dir = path.join(AI_DIR, "2026-09-26", "health-taskfor-landed");
const file = "public/framework/ux/health-taskfor-proof.js";
fs.mkdirSync(old_dir, { recursive: true }); fs.mkdirSync(landed_dir, { recursive: true });
const line = o => JSON.stringify(o) + "\n";
fs.writeFileSync(path.join(old_dir, "task.jsonl"), line({ assign: { session_id: "proof", steps: ["a"], step: 0 } }) + line({ action: { at: "2026-09-25T10:00:00-04:00", did: "edit", files: [file] } }));
fs.writeFileSync(path.join(landed_dir, "task.jsonl"), line({ assign: { session_id: "proof", steps: ["a"], step: 0 } }) + line({ action: { at: "2026-09-26T10:00:00-04:00", did: "edit", files: [file] } }) + line({ assign: { step: 1, landed_at: "2026-09-26T11:00:00-04:00", outcome: "x" } }));
try {
	let t0 = performance.now(); const cold = task_for(file); const cold_ms = performance.now() - t0;
	t0 = performance.now(); const warm = task_for(file); const warm_ms = performance.now() - t0;
	console.log("file edited by a task opened 2026-09-25, a newer landed task also lists it");
	console.log("task_for ->", cold && path.relative(ROOT, cold).replaceAll("\\", "/"), `(cold ${cold_ms.toFixed(0)} ms, warm ${warm_ms.toFixed(0)} ms)`);
	console.log("unknown file ->", task_for("public/framework/ux/nobody-edited-this.js"));

	for (const d of ["2026-09-25", "2026-09-26"]) { const p = path.join(AI_DIR, d); if (fs.existsSync(p) && !fs.readdirSync(p).length) fs.rmdirSync(p); }
}
